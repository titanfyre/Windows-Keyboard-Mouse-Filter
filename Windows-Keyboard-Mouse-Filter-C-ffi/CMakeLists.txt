cmake_minimum_required(VERSION 3.24)
project(WindowsKeyboardMouseFilterBridge VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FAMILY_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical project SDK source")
if(NOT EXISTS "${FAMILY_NATIVE_SOURCE}/CMakeLists.txt")
  set(FAMILY_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Windows-Keyboard-Mouse-Filter-CPP-vcpkg")
endif()
option(FAMILY_TEST_BACKEND "Synthetic backend; never opens a physical driver" OFF)
add_library(WindowsKeyboardMouseFilterBridge SHARED Bridge.cpp)
target_include_directories(WindowsKeyboardMouseFilterBridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${FAMILY_NATIVE_SOURCE}/sdk")
target_compile_features(WindowsKeyboardMouseFilterBridge PRIVATE cxx_std_17)
target_compile_definitions(WindowsKeyboardMouseFilterBridge PRIVATE IC_BRIDGE_EXPORTS UNICODE _UNICODE NOMINMAX)
target_compile_options(WindowsKeyboardMouseFilterBridge PRIVATE /W4 /EHsc /utf-8)
if(FAMILY_TEST_BACKEND)
 target_sources(WindowsKeyboardMouseFilterBridge PRIVATE tests/FakeSdk.cpp)
 enable_testing()
 add_executable(family_contract tests/Contract.cpp)
 target_compile_definitions(family_contract PRIVATE TEST_INPUT_MASK=3u TEST_CONTROLLER_MASK=0u TEST_PRODUCT="Windows-Keyboard-Mouse-Filter")
 target_link_libraries(family_contract PRIVATE WindowsKeyboardMouseFilterBridge)
 add_test(NAME family_contract COMMAND family_contract)
else()
 add_subdirectory("${FAMILY_NATIVE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/native")
 target_link_libraries(WindowsKeyboardMouseFilterBridge PRIVATE WindowsKeyboardMouseFilterSDK::sdk)
endif()
